cmake_minimum_required(VERSION 3.25)
project(WindowsUnlock VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
  URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
FetchContent_MakeAvailable(json)
add_library(phoneunlock_core core/core.cpp core/http.cpp core/remote.cpp core/camera.cpp)
target_include_directories(phoneunlock_core PUBLIC core)
target_link_libraries(phoneunlock_core PUBLIC nlohmann_json::nlohmann_json bcrypt ncrypt crypt32 winhttp ole32 mf mfplat mfreadwrite mfuuid windowscodecs powrprof advapi32 user32)
target_compile_definitions(phoneunlock_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(phoneunlock_core PRIVATE /W4 /permissive- /utf-8 /guard:cf)
add_executable(phoneunlock WIN32 desktop-test-client/main.cpp)
target_link_libraries(phoneunlock PRIVATE phoneunlock_core comctl32 shell32 wtsapi32)
target_link_options(phoneunlock PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE phoneunlock_core)
add_executable(http_tests tests/http_tests.cpp)
target_link_libraries(http_tests PRIVATE phoneunlock_core)
enable_testing()
add_test(NAME core_security COMMAND core_tests)

# Development native integration. No registration/install action is part of a build.
add_library(phoneunlock_ipc STATIC native/ipc.cpp)
target_include_directories(phoneunlock_ipc PUBLIC native)
target_compile_definitions(phoneunlock_ipc PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(phoneunlock_ipc PRIVATE /W4 /permissive- /utf-8 /guard:cf)
target_link_libraries(phoneunlock_ipc PUBLIC advapi32)
add_library(CredentialProviderPreview SHARED credential-provider/provider.cpp credential-provider/exports.def)
target_link_libraries(CredentialProviderPreview PRIVATE phoneunlock_ipc ole32 uuid gdi32)
target_compile_options(CredentialProviderPreview PRIVATE /W4 /permissive- /utf-8 /guard:cf)
target_link_options(CredentialProviderPreview PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf /CETCOMPAT)
add_library(phoneunlock_preview_broker STATIC service/storage.cpp service/broker.cpp)
target_include_directories(phoneunlock_preview_broker PUBLIC service)
target_link_libraries(phoneunlock_preview_broker PUBLIC phoneunlock_core phoneunlock_ipc shell32 wtsapi32)
target_compile_options(phoneunlock_preview_broker PRIVATE /W4 /permissive- /utf-8 /guard:cf)
add_executable(PhoneUnlockPreviewService service/main.cpp)
target_link_libraries(PhoneUnlockPreviewService PRIVATE phoneunlock_preview_broker)
add_executable(PhoneUnlockPreviewStage service/stage.cpp)
target_link_libraries(PhoneUnlockPreviewStage PRIVATE phoneunlock_preview_broker)
foreach(target PhoneUnlockPreviewService PhoneUnlockPreviewStage)
  target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /guard:cf)
  target_link_options(${target} PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf /CETCOMPAT)
endforeach()
add_executable(native_tests tests/native_tests.cpp)
target_link_libraries(native_tests PRIVATE phoneunlock_ipc)
add_executable(provider_tests tests/provider_tests.cpp)
target_link_libraries(provider_tests PRIVATE phoneunlock_ipc ole32 uuid gdi32)
foreach(target native_tests provider_tests)
  target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /guard:cf)
  target_link_options(${target} PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
endforeach()
add_dependencies(provider_tests CredentialProviderPreview)
add_test(NAME native_ipc_security COMMAND native_tests)
add_test(NAME provider_com_security COMMAND provider_tests $<TARGET_FILE:CredentialProviderPreview>)
add_executable(enrollment_tests tests/enrollment_tests.cpp)
target_link_libraries(enrollment_tests PRIVATE phoneunlock_preview_broker)
target_compile_options(enrollment_tests PRIVATE /W4 /permissive- /utf-8 /guard:cf)
add_test(NAME preview_enrollment_security COMMAND enrollment_tests)
