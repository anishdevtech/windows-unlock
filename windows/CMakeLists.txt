cmake_minimum_required(VERSION 3.25)
project(WindowsUnlock VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
  URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
FetchContent_MakeAvailable(json)
add_library(phoneunlock_crypto core/core.cpp)
target_include_directories(phoneunlock_crypto PUBLIC core)
target_link_libraries(phoneunlock_crypto PUBLIC nlohmann_json::nlohmann_json bcrypt ncrypt crypt32 ole32)
target_compile_definitions(phoneunlock_crypto PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(phoneunlock_crypto PRIVATE /W4 /permissive- /utf-8 /guard:cf)
add_library(phoneunlock_core core/http.cpp core/remote.cpp core/camera.cpp core/telemetry.cpp)
target_include_directories(phoneunlock_core PUBLIC core)
target_link_libraries(phoneunlock_core PUBLIC phoneunlock_crypto winhttp mf mfplat mfreadwrite mfuuid windowscodecs powrprof advapi32 user32)
target_compile_definitions(phoneunlock_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(phoneunlock_core PRIVATE /W4 /permissive- /utf-8 /guard:cf)
add_library(phoneunlock_qr STATIC third-party/qrcodegen/qrcodegen.cpp)
target_include_directories(phoneunlock_qr PUBLIC third-party/qrcodegen)
add_executable(phoneunlock WIN32 desktop-test-client/main.cpp desktop-test-client/pairing_qr.cpp)
target_compile_options(phoneunlock PRIVATE /W4 /permissive- /utf-8 /guard:cf)
target_link_libraries(phoneunlock PRIVATE phoneunlock_core phoneunlock_qr comctl32 shell32 wtsapi32 dwmapi)
target_link_options(phoneunlock PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE phoneunlock_core)
add_executable(http_tests tests/http_tests.cpp)
target_link_libraries(http_tests PRIVATE phoneunlock_core)
enable_testing()
add_test(NAME core_security COMMAND core_tests)
